#!/usr/bin/env python3 # -*- coding: utf-8 -*- """ داشبورد متابعة الأسهم - الكود الوظيفي فقط أسهم التقسيم العكسي الإيجابية """ import json import os from datetime import datetime from typing import List, Dict, Optional DATA_FILE = "dashboard_data.json" # ========================= # تحميل وحفظ البيانات # ========================= def load_data() -> Dict: """تحميل البيانات من ملف JSON""" if not os.path.exists(DATA_FILE): return { "updated_at": None, "stats": { "watch": 0, "opportunities": 0, "signals": 0 }, "stocks": [] } with open(DATA_FILE, "r", encoding="utf-8") as f: return json.load(f) def save_data(data: Dict) -> None: """حفظ البيانات في ملف JSON""" data["updated_at"] = datetime.now().strftime("%Y-%m-%d %H:%M") with open(DATA_FILE, "w", encoding="utf-8") as f: json.dump(data, f, ensure_ascii=False, indent=2) # ========================= # إدارة الأسهم # ========================= def add_stock( symbol: str, name: str = "", status: str = "watch", watchlist: bool = False ) -> None: """ إضافة سهم جديد status: watch | candidate | ready """ data = load_data() # منع التكرار for s in data["stocks"]: if s["symbol"].upper() == symbol.upper(): print(f"السهم {symbol} موجود مسبقاً") return data["stocks"].append({ "symbol": symbol.upper(), "name": name, "status": status, "watchlist": watchlist }) update_stats(data) save_data(data) print(f"تمت إضافة {symbol}") def remove_stock(symbol: str) -> None: """حذف سهم""" data = load_data() before = len(data["stocks"]) data["stocks"] = [s for s in data["stocks"] if s["symbol"].upper() != symbol.upper()] if len(data["stocks"]) == before: print(f"السهم {symbol} غير موجود") return update_stats(data) save_data(data) print(f"تم حذف {symbol}") def update_stock_status(symbol: str, status: str) -> None: """تحديث حالة السهم: watch | candidate | ready""" data = load_data() found = False for s in data["stocks"]: if s["symbol"].upper() == symbol.upper(): s["status"] = status found = True break if not found: print(f"السهم {symbol} غير موجود") return update_stats(data) save_data(data) print(f"تم تحديث حالة {symbol} إلى {status}") def toggle_watchlist(symbol: str) -> None: """إضافة/إزالة من قائمة المتابعة""" data = load_data() found = False for s in data["stocks"]: if s["symbol"].upper() == symbol.upper(): s["watchlist"] = not s.get("watchlist", False) found = True print(f"{symbol} → قائمة المتابعة: {s['watchlist']}") break if not found: print(f"السهم {symbol} غير موجود") return save_data(data) # ========================= # الإحصائيات والفلترة # ========================= def update_stats(data: Dict) -> None: """حساب الإحصائيات تلقائياً""" stocks = data["stocks"] data["stats"] = { "watch": len(stocks), "opportunities": len([s for s in stocks if s.get("status") == "candidate"]), "signals": len([s for s in stocks if s.get("status") == "ready"]) } def get_stats() -> Dict: """إرجاع الإحصائيات الحالية""" data = load_data() return data.get("stats", {}) def filter_stocks( status: Optional[str] = None, watchlist_only: bool = False, search: str = "" ) -> List[Dict]: """ فلترة الأسهم status: None | watch | candidate | ready """ data = load_data() result = data["stocks"] if status: result = [s for s in result if s.get("status") == status] if watchlist_only: result = [s for s in result if s.get("watchlist")] if search: q = search.lower() result = [ s for s in result if q in s.get("symbol", "").lower() or q in s.get("name", "").lower() ] return result # ========================= # عرض سريع في التيرمينال # ========================= def show_dashboard() -> None: """عرض ملخص الداشبورد في التيرمينال""" data = load_data() stats = data.get("stats", {}) print("\n" + "=" * 40) print(" داشبورد متابعة الأسهم") print("=" * 40) print(f" أسهم تحت المراقبة : {stats.get('watch', 0)}") print(f" فرص حالية : {stats.get('opportunities', 0)}") print(f" إشارات جاهزة : {stats.get('signals', 0)}") print(f" آخر تحديث : {data.get('updated_at') or '—'}") print("=" * 40) stocks = data.get("stocks", []) if not stocks: print(" لا توجد أسهم حالياً") return print("\n الأسهم:") for s in stocks: status_map = { "ready": "جاهز للدخول", "candidate": "مرشح للصعود", "watch": "تحت المراقبة" } status_ar = status_map.get(s.get("status"), s.get("status")) watch = "★" if s.get("watchlist") else " " print(f" {watch} {s['symbol']:<8} | {status_ar:<15} | {s.get('name', '')}") # ========================= # تشغيل مباشر # ========================= if __name__ == "__main__": # أمثلة استخدام: # إضافة أسهم # add_stock("AAPL", "Apple Inc.", status="ready", watchlist=True) # add_stock("TSLA", "Tesla", status="candidate") # add_stock("NVDA", "NVIDIA", status="watch") # تحديث حالة # update_stock_status("TSLA", "ready") # فلترة # print(filter_stocks(status="ready")) # print(filter_stocks(watchlist_only=True)) # print(filter_stocks(search="app")) # عرض الداشبورد show_dashboard()
